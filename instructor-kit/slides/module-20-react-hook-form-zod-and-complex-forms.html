8.2 · Counts

Check the text, then a number

shoreline-hq/src/api/schemas.ts
function wholeNumber(max: number, message: string) {
  return z
    .string()
    .trim()
    .regex(/^\d+$/, message)
    .transform(Number)
    .refine((count) => count <= max, message)
}
WHY Input string, output number: the same split as capacity.
