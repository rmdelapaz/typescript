src/components/PlotBoard.tsx, the derived list and count:

import { useState } from 'react'
import { plots } from '../data/plots.ts'
import { countLine, filterPlots, type BedFilter, type StatusFilter } from '../utils/labels.ts'
import styles from './PlotBoard.module.css'
import { PlotCard } from './PlotCard.tsx'
import { PlotFilters } from './PlotFilters.tsx'

export function PlotBoard() {
  const [status, setStatus] = useState<StatusFilter>('all')
  const [bed, setBed] = useState<BedFilter>('all')

  // Derived on every render, never stored: the list and its count always agree with the filters
  const visible = filterPlots(plots, status, bed)

  function clearFilters() {
    setStatus('all')
    setBed('all')
  }

  return (
    <>
      <PlotFilters
        status={status}
        bed={bed}
        onStatusChange={setStatus}
        onBedChange={setBed}
        onClear={clearFilters}
      />
      <p className={styles.count} role="status">
        {countLine(visible.length, plots.length)}
      </p>
      {visible.length === 0 ? (
        <p className={styles.empty}>No plots match these filters.</p>
      ) : (
        <ul className={styles.list}>
          {visible.map((plot) => (
            <li key={plot.id}>
              <PlotCard plot={plot} />
            </li>
          ))}
        </ul>
      )}
    </>
  )
}

  emptyValues,
  readValues,
  validateRequest,
  type FieldErrors,
  type RequestField,
  type RequestValues,
} from '../utils/requestForm.ts'
import buttonStyles from './Button.module.css'
import { FieldError } from './FieldError.tsx'
import formStyles from './Form.module.css'

// What the form shows: what the person typed and what to fix, or the request it received
type FormState =
  | { kind: 'editing'; values: RequestValues; errors: FieldErrors }
  | { kind: 'sent'; request: PlotRequest }

const initialState: FormState = { kind: 'editing', values: emptyValues, errors: {} }

// The form's action: React calls it with the previous state and the submitted FormData
function sendRequest(_previous: FormState, formData: FormData): FormState {
  const values = readValues(formData)
  const errors = validateRequest(values)
  // isBedKind narrows the text to a BedKind (validateRequest has already said it is one)
  if (Object.keys(errors).length > 0 || !isBedKind(values.bed)) {
    return { kind: 'editing', values, errors }
  }
  return {
    kind: 'sent',
    request: {
      name: values.name,
      email: values.email,
      bed: values.bed,
      householdSize: Number(values.householdSize),
    },
  }
}

export function RequestForm() {
  const [state, formAction] = useActionState(sendRequest, initialState)
  const values = state.kind === 'editing' ? state.values : emptyValues
  const errors = state.kind === 'editing' ? state.errors : {}

  // The props every field needs: its name, a link to its message, and whether it is invalid
  function fieldProps(field: RequestField) {
    const invalid = errors[field] !== undefined
    return {
      id: field,
      name: field,
      'aria-invalid': invalid,
      'aria-describedby': invalid ? `${field}-error` : undefined,
    }
  }
  // ... the fields, each with its label, defaultValue and FieldError (see solution/)
}

Written part: questions, model answers, marking
W1
A teammate runs node plots.ts on a file that has a type error, and Node prints a result. Why does Node not complain, and which command finds the error? (1)
Node removes the types and runs the JavaScript that is left: it never checks them. npx tsc --noEmit (here npm run typecheck, or npm run build, which runs tsc first) reads the types and reports the error.
Marking: The point needs both halves: running is not checking, and a tsc command.
W2
A learner shows the gardener's name in capitals and tsc prints the error below. What does it mean, and how do you fix it? (2)
src/components/PlotCard.tsx:26:77 - error TS18048: 'plot.gardener' is possibly 'undefined'.

W4
Why does the plot list use key={plot.id} and not the array index? (1)
A key tells React which item is which between renders. When the filters remove or reorder plots, an index key would point at a different plot, so state such as an open details toggle would jump to the wrong card. An id stays with its plot.
Marking: Accept any answer that ties the key to the item's identity when the list changes.
W5

The count can be worked out from state on every render: storing it adds a second render and a moment where the number and the list disagree (and visible is a new array each render, so the effect runs every time). Write const shown = visible.length (derived, never stored).
Marking: 1 point for "derived value kept in state", 1 for the one-line fix.

src/auth/guards.ts: the guard loader (redirect with ?next= for guests, 403 for staff) and nextPath:

// Route guards: loaders that decide who may open a page, before the page renders (Lesson 6.3)
import { data, redirect, type LoaderFunctionArgs } from 'react-router'
import { readSession } from './session.ts'

/**
 * Where to go after signing in: the ?next= address, but only if it is a path inside this app.
 * Anything else (https://elsewhere.example, //elsewhere.example) goes to the home page.
 */
export function nextPath(url: URL): string {
  const next = url.searchParams.get('next')
  if (next === null || !next.startsWith('/') || next.startsWith('//') || next.startsWith('/\\')) {
    return '/'
  }
  return next
}

/** For the setters' pages: guests sign in first, staff are turned away */
export function settersOnly({ request }: LoaderFunctionArgs): null {
  const session = readSession()
  if (session === null) {
    // Remember the page (and its ?search) so the sign-in page can send people back to it
    const { pathname, search } = new URL(request.url)
    throw redirect(`/sign-in?next=${encodeURIComponent(pathname + search)}`)
  }
  if (session.user.role !== 'setter') {
    const message = 'Your staff account can read every route, but only setters can log new ones. Ask Dana.'
    throw data(message, { status: 403, statusText: 'Only setters can log routes' })
  }
  return null
}

src/state/routeDraft.ts: the action union and the reducer, ending in assertNever:

/** Every change the form can make: one object type per kind of change */
export type DraftAction =
  | { type: 'field'; field: DraftField; value: string }
  | { type: 'submitted' }
  | { type: 'failed'; message: string; fieldErrors: Record<string, string[]> }
  | { type: 'reset' }

export const emptyDraft: RouteDraft = {
  values: { name: '', wallId: '', grade: '', holdColor: '' },
  status: 'editing',
  fieldErrors: {},
  message: '',
}

const draftFields: readonly DraftField[] = ['name', 'wallId', 'grade', 'holdColor']

export function routeDraftReducer(state: RouteDraft, action: DraftAction): RouteDraft {
  switch (action.type) {
    case 'field': {
      // Typing in a field takes its old message away; the other fields keep theirs
      const { [action.field]: _fixed, ...otherErrors } = state.fieldErrors
      return {
        ...state,
        values: { ...state.values, [action.field]: action.value },
        fieldErrors: otherErrors,
      }
    }
    case 'submitted':
      return { ...state, status: 'sending', message: '' }
    case 'failed': {
      // Keep only the messages for fields this form has
      const fieldErrors: RouteDraft['fieldErrors'] = {}
      for (const field of draftFields) {
        const messages = action.fieldErrors[field]
        if (messages !== undefined) fieldErrors[field] = messages
      }
      return { ...state, status: 'editing', message: action.message, fieldErrors }
    }
    case 'reset':
      return emptyDraft
    default:
      return assertNever(action)
  }
}
